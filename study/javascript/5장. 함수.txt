5장. 함수
- 익명 함수(함수 표현식, 화살표 함수), 선언적 함수, 매개변수와 리턴값
- 가변 인자 함수(arguments), 내부 함수(스코프 격리/충돌 방지), 콜백 함수, setTimeout(), setInterval()


1. 익명 함수
- 구문 : var 함수명 = function() {}; / var 함수명 = () => {};

- 이름이 없는 함수로, 변수에 할당하여 사용 (함수 표현식)
- ES6 화살표 함수(Arrow Function) 문법 지원

  * 화살표 함수 문법 규칙
    - 매개변수가 없을 때 : () => { ... }
    - 매개변수가 1개일 때 : x => { ... }
    - 매개변수가 2개 이상일 때 : (x, y) => { ... }
    - 몸체가 단일 return문일 때 : x => x * x (중괄호 및 return 생략 가능)
    - 객체를 직접 반환할 때 : () => ({ a: 1 })
    - 여러 줄 구현부 : () => { const x = 10; return x * x; }

	ex) 익명 함수
		<script>
			let functionVar = function(i, j) {
				alert(`함수 표현식 test i=${i}, j=${j}`);
			}; 
			functionVar(10, 20); // 호출

			// 화살표 함수 재할당
			functionVar = (i, j) => {
				alert(i + j);
			};
		</script>

	ex) 화살표 함수의 다양한 형태
		let funVar = function() {
			console.log('1. 일반함수 호출');
		};
		funVar();

		funVar = () => console.log('2. 매개변수가 없거나 2개 이상인 화살표함수 호출');
		funVar();

		funVar = i => {
			console.log('3. 매개변수가 하나인 화살표 함수 호출');
			console.log('\t매개변수 i = ', i);
		};
		funVar(10);

		funVar = i => console.log('4. 매개변수 하나, 한 줄 구현부', i);
		funVar(10);

		funVar = i => i * i;
		console.log('5. 매개변수 하나, 단일 리턴값:', funVar(10));

		funVar = (i, j) => i + j;
		console.log('6. 매개변수 2개 이상, 단일 리턴값:', funVar(10, 20));


2. 선언적 함수
- 구문 :  function 함수명 () {};

- 함수 선언부보다 상단에서 호출 가능
- 동일한 이름으로 재선언 시 마지막에 선언된 함수로 덮어씌워짐(전역 적용)

	ex) 선언적 함수
		<script>
			fun(); // 정의 전 호출 가능 (수정된 함수 fun 출력)

			function fun() {
				document.write('함수 fun <br>');
			}

			function fun() { // 기존 선언을 덮어씀
				document.write('수정된 함수 fun <br>');
			}
			fun();
		</script>


3. 매개변수와 리턴값

- 선언된 매개변수 개수와 호출 시 전달하는 인자 개수가 일치하지 않아도 오류가 발생하지 않음
	* 인자가 더 많은 경우 : 앞선 매개변수 순서대로 할당되고 남는 인자는 무시
	* 인자가 부족한 경우 : 할당되지 않은 매개변수는 undefined 값을 가짐 (연산 시 NaN 결과)

- return 문이 없거나 값이 없으면 undefined 반환

	ex) 기본값을 갖는 매개변수
	console.log(pow(5,3));
	console.log(pow(5));
	console.log(pow(y=2, x=3)); // 첫번째 매개변수 x(2), 두번째 매개변수 y(3) ; 
			// 서순 지정 불가
	console.log(pow());

	function pow(x=5, y=2) {
    		console.log(`함수 내의 x=${x}, y=$${y}`);
    		// x^y를 return
    		result = x ** y;
    		return result;}

	ex)인자 개수 불일치 처리
		function pow(x, y) {
			console.log(`x=${x}, y=${y}`);
			return x ** y;
		}

		console.log(pow(5, 3));      // 정상 호출 -> 125
		console.log(pow(5, 3, 2));   // 초과 인자 무시 -> 125
		console.log(pow());          // 부족한 인자 (x,y=undefined) -> NaN


4. 가변 인자 함수

- 매개변수의 개수가 유동적으로 변할 수 있는 함수
- arguments : 함수 내부에서 전달된 모든 인자를 담고 있는 유사 배열 객체 (화살표 함수에서 arguments 사용 불가)

	ex )Array() 함수
		function array() {
			let result = [];
			if (arguments.length === 1) {
				// 인자가 1개이고 숫자인 경우 : 해당 길이만큼의 배열 생성
				for (let cnt = 1; cnt <= arguments[0]; cnt++) {
					result.push(undefined);
				}
			} else if (arguments.length >= 2) {
				// 인자가 2개 이상인 경우 : 인자들을 요소로 채움
				for (let idx = 0; idx < arguments.length; idx++) {
					result.push(arguments[idx]);
				}
			}
			return result;
		}


5. array

- array 함수 : 가변인자함수 (함수표현식 활용)
 	* 매개변수 0개 : 빈 배열 [] 생성 (length가 0인 배열)
 	* 매개변수 1개 : 해당 숫자를 길이(length)로 갖는 빈 희소 배열 생성
 	* 매개변수 2개 이상 : 전달받은 모든 인자들을 요소로 포함하는 배열 생성

	ex) array()를 Array()로 작동하도록 구현
	function array() { // 가변인자함수에서, 매개변수는 지정하지 않음
    		let result = [];
    		if(arguments.length==1){
        		for(let cnt=1 ; cnt<=arguments[0] ; cnt ++){
            			result.push(undefined);
        		}// for
   		} else if(arguments.length>=2){
        		for(let data of arguments){
            			result.push(data);
       			}//for
    		}//if
    	return result;};

	// array()와 Array()의 다른 부분
	var arr4 = array(3); // [ undefined, undefined, undefined ]
	var arr5 = Array(3); // [ <3 empty items> ]

- 전역 공간에서의 함수/변수 이름 충돌을 방지함

            		let square = x => x*x;
            		return Math.sqrt(square(width) + square(height));}

        	function square(width, height, hypotenuse) {
            		return width ** 2 + height ** 2 == hypotenuse ** 2;}
   	 </script>


7. 콜백 함수

- 다른 함수의 매개변수로 전달되어 특정 시점에 실행되는 함수
- 익명 함수나 화살표 함수 형태로 직접 전달하여 많이 활용됨

	ex) 콜백 함수 전달
		<script>
       		 // 콜백함수 (매개변수로 들어온 함수)
        		function calltentimes (callback) {
            			for(var i=0 ; i<10 ; i++){
                			callback();
            			}
        		}

        		let fun = () => document.write(new Date().toLocaleString()+'<br>');
        		calltentimes(fun);

        		calltentimes(function(){
            			document.write(new Date().toLocaleString()+'<br>')
        		})

        		calltentimes(() => document.write(new Date().toLocaleString()+'<br>'));
		</script>


8. 타이머 내장 함수
- 시간 관련 비동기 처리를 담당하는 내장 함수

  1. 변수 = setTimeout(function, millisecond) : 지정 시간(ms) 후 함수를 1회 실행
  2. 변수 = setInterval(function, millisecond) : 지정 시간(ms) 마다 함수를 반복 실행
  3. clearTimeout(변수) : 설정된 setTimeout 타이머 취소
  4. clearInterval(변수) : 설정된 setInterval 타이머 취소

	ex) 1초마다 배경이 바뀌며 10초 후 종료
	<script>
        	// img01 : 노란색, img02 : 파란색, img03 : 하얀색
        	let sec = 0; // 시간 경과 저장 변수
        	document.write(`<p>${new Date().toLocaleString()} - ${sec}초 경과<br>${10-sec}초</p>`);

        	var img = [
        		'url(ch05_img/img01.png)',
        		'url(ch05_img/img02.png)',
        		'url(ch05_img/img03.png)'
        		]

        	var color = [
            		'yellow',
            		'blue',
           		'white'
        		];

        	let id = setInterval(() => {
            		sec++;
            		document.body.innerHTML = `<p>${new Date().toLocaleString()} - ${sec}초 경과<br>${10-sec}초</p>`;
            		document.body.style.backgroundImage = img[sec%3];
            		document.body.style.color = color[sec%3];
        	}, 1000);

        	setTimeout(() => {
            		clearInterval(id);            
        	}, 10000);
        
        	setTimeout(() => {
            		document.body.innerHTML = `<p>종료</p>`;
            		document.body.style.color = 'white';
            		document.body.style.backgroundImage = 'none';
            		document.body.style.backgroundColor = 'black';
        	}, 11000);
    	</script>
