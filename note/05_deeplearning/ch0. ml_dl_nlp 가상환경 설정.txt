Conda 가상환경 구축 및 NLP 개발 환경 가이드


# 1. 가상환경 생성 및 활성화

# 기존 가상환경 목록 확인
conda env list

# 신규 가상환경 생성 (Python 3.10 기준)
conda create -n ml-dl-nlp python=3.10 -y

# 가상환경 활성화
conda activate ml-dl-nlp


# 2. Jupyter Notebook 커널 등록 및 연결

# 가상환경에 ipykernel 설치
conda install ipykernel -y

# Jupyter Notebook에 커널 연결
python -m ipykernel install --user --name ml-dl-nlp --display-name "ml-dl-nlp(ipykernel)"

# 등록된 커널 목록 확인
jupyter kernelspec list


# 3. 패키지 관리자 및 필수 라이브러리 설치

# pip 최신 버전 업그레이드
python -m pip install --upgrade pip

# 데이터 분석, 머신러닝 및 딥러닝 패키지
pip install jupyter numpy==1.23.5 pandas==1.5.3 matplotlib==3.7.0 seaborn==0.12.2 scikit-learn==1.2.1 tensorflow==2.10

# 데이터 및 파일 관리 (Excel, Env)
pip install openpyxl==3.1.5 python-dotenv

# 컴퓨터 비전 (CV)
pip install opencv-python==4.8.1.78

# 자연어 처리 (NLP) 및 시각화
pip install nltk==3.7 konlpy python-mecab-ko gensim wordcloud

# 시계열 분석 및 연관 분석
pip install apyori statsmodels==0.13.5 prophet plotly


# 4. 커널 삭제 및 가상환경 제거 (초기화 필요 시)

# Jupyter 커널 삭제 (--name으로 지정했던 실제 커널명 입력)
jupyter kernelspec remove ml-dl-nlp

# Conda 가상환경 완전 삭제
conda env remove -n ml-dl-nlp
